\[
  \operatorname{Var}\!\Big(\frac{q^\top k}{\sqrt{d_k}}\Big)
  = \frac{1}{d_k}\sum_{i=1}^{d_k}\operatorname{Var}(q_i k_i)
  = \frac{1}{d_k}\cdot d_k = 1
\]
